**Goal:** Make the scraper stream one profile at a time (scrape → email → verify → write → next), remember seen profiles across runs globally, and expand search criteria automatically to hit arbitrarily large `max_results`.

**Architecture:** A new `SeenProfiles` class provides file-backed global dedup. `search_people` becomes a generator `search_people_stream` that expands criteria and yields one domain-resolved profile at a time. `run_pipeline` accepts any iterable and gains `log`/`on_profile_done` callbacks, writing each result immediately. The runner and UI wire it all together.

**Tech Stack:** Python 3.14, Playwright (sync), pytest, pytest-mock, Streamlit 1.44, Click

```python
import os


class SeenProfiles:
    """File-backed set of LinkedIn profile IDs, persisted across runs."""

    def __init__(self, path: str) -> None:
        self._path = path
        self._ids: set[str] = self._load()

    def _load(self) -> set[str]:
        if not os.path.exists(self._path):
            return set()
        with open(self._path, encoding="utf-8") as f:
            return {line.strip() for line in f if line.strip()}

    def __contains__(self, profile_id: str) -> bool:
        return profile_id in self._ids

    def add(self, profile_id: str) -> None:
        if profile_id not in self._ids:
            self._ids.add(profile_id)
            os.makedirs(os.path.dirname(self._path) or ".", exist_ok=True)
            with open(self._path, "a", encoding="utf-8") as f:
                f.write(profile_id + "\n")

    def __len__(self) -> int:
        return len(self._ids)
```

```python
]


def _expand_criteria(criteria: str) -> list[str]:
    """Return original + related search terms for large-scale scraping."""
    base = criteria.strip()
    results = [base]
    seen_lower: set[str] = {base.lower()}
    lowered = base.lower()

    for keyword, synonyms in _CRITERIA_EXPANSIONS.items():
        if keyword in lowered:
            for s in synonyms:
                if s.lower() not in seen_lower:
                    seen_lower.add(s.lower())
                    results.append(s)

    for suffix in _GENERIC_SUFFIXES:
        candidate = f"{base} {suffix}"
        if candidate.lower() not in seen_lower:
            seen_lower.add(candidate.lower())
            results.append(candidate)

    return results
```

```python
def search_people_stream(
    context: BrowserContext,
    criteria: str,
    max_results: int = 100,
    seen_ids=None,
) -> Iterator[Profile]:
    """
    Generator yielding one domain-resolved Profile at a time.
    Expands criteria automatically and skips profiles in seen_ids.
    Continues across multiple search terms until max_results is reached.
    """
    if seen_ids is None:
        seen_ids = set()

    yielded = 0
    local_seen: set[str] = set()
    tried: set[str] = set()

    for current_criteria in _expand_criteria(criteria):
        if yielded >= max_results:
            return
        key = current_criteria.lower()
        if key in tried:
            continue
        tried.add(key)

        page_num = 1
        page = context.new_page()
        try:
            while yielded < max_results:
                url = build_people_search_url(current_criteria, page_num)
                page.goto(url, wait_until="domcontentloaded", timeout=20_000)
                _wait_for_search_results(page)
                time.sleep(1)

                if _is_login_page(page):
                    return

                cards = _get_result_cards(page)
                if not cards:
                    break

                batch = extract_profiles_from_page(cards)
                if not batch:
                    break

                fresh = [
                    p for p in batch
                    if p.profile_id
                    and p.profile_id not in seen_ids
                    and p.profile_id not in local_seen
                ]

                for profile in fresh:
                    if yielded >= max_results:
                        break
                    local_seen.add(profile.profile_id)

                    if profile.company_name and not profile.company_domain:
                        slug = (
                            profile.company_name.lower()
                            .replace(" ", "-")
                            .replace(",", "")
                        )
                        profile.company_domain = resolve_company_domain(
                            context,
                            f"https://www.linkedin.com/company/{slug}",
                            company_name=profile.company_name,
                        )
                        time.sleep(0.5)

                    yield profile
                    yielded += 1

                next_btn = page.query_selector(
                    'button[aria-label="Next"]'
                ) or page.query_selector(
                    "button.artdeco-pagination__button--next"
                )
                if not next_btn or not next_btn.is_enabled():
                    break

                page_num += 1
                time.sleep(1.5)
        finally:
            page.close()
```

```python
import sys
from collections.abc import Callable, Iterable

from src.email_generator import generate_emails
from src.email_pattern_detector import get_domain_pattern
from src.models import EmailResult, Profile
from src.sheets_writer import CSVWriter
from src.smtp_verifier import verify_email, verify_first_valid


def run_pipeline(
    profiles: Iterable[Profile],
    context,
    writer: CSVWriter,
    log: Callable[[str], None] | None = None,
    on_profile_done: Callable[[str], None] | None = None,
) -> tuple[int, int, int]:
    """
    Process each profile: pattern detect → generate → SMTP verify → write.
    Processes one profile at a time as they arrive from the iterable.
    Returns (verified_count, rejected_count, skipped_count).
    """
    logger = log or (lambda _: None)
    mark_done = on_profile_done or (lambda _: None)

    verified_count = rejected_count = skipped_count = 0
    pattern_cache: dict = {}
    catch_all_cache: dict = {}
    total = 0

    for profile in profiles:
        total += 1
        try:
            domain = profile.company_domain

            if not domain:
                result = EmailResult(profile=profile, email=None, status="no_domain")
                writer.write_rejected(result)
                rejected_count += 1
                logger(
                    f"[{total}] {profile.full_name} @ {profile.company_name or '?'}"
                    f" → no domain"
                )
                mark_done(profile.profile_id)
                continue

            if domain not in pattern_cache:
                pattern_cache[domain] = get_domain_pattern(context, domain)
            pattern = pattern_cache[domain]

            emails = generate_emails(profile, pattern=pattern, domain=domain)

            if not emails:
                result = EmailResult(profile=profile, email=None, status="no_pattern")
                writer.write_rejected(result)
                rejected_count += 1
                logger(
                    f"[{total}] {profile.full_name} @ {profile.company_name or '?'}"
                    f" → no email pattern"
                )
                mark_done(profile.profile_id)
                continue

            if len(emails) == 1:
                status = verify_email(emails[0], catch_all_cache)
                email = emails[0]
            else:
                email, status = verify_first_valid(emails, catch_all_cache)

            result = EmailResult(profile=profile, email=email, status=status)

            if result.goes_to_verified:
                writer.write_verified(result)
                verified_count += 1
            else:
                writer.write_rejected(result)
                rejected_count += 1

            email_display = email or "—"
            logger(
                f"[{total}] {profile.full_name} @ {profile.company_name or '?'}"
                f" → {email_display} ({status})"
            )
            mark_done(profile.profile_id)

        except Exception as exc:
            print(f"[pipeline] Skipping {profile.full_name}: {exc}", file=sys.stderr)
            skipped_count += 1
            logger(f"[{total}] {profile.full_name} → error: {exc}")

    return verified_count, rejected_count, skipped_count
```

```python
def _read_csv_rows(path: str) -> list[dict[str, str]]:
    if not os.path.exists(path):

def run_outreach(

    context = get_authenticated_context(cookies_file)

    try:
        seen = SeenProfiles(seen_profiles_file)
        logger(f"      {len(seen)} profiles already seen from previous runs.")

        writer = CSVWriter(output_dir=output_dir)
        logger(f"[2/3] Searching for '{criteria}' (target: {max_results})...")
        logger(f"      Output: {output_dir}/  |  Seen: {seen_profiles_file}")

        profiles_stream = search_people_stream(

            max_results=max_results,
            seen_ids=seen,
        )

        logger("[3/3] Processing profiles one by one (scrape → email → verify → write)...")

        verified, rejected, skipped = run_pipeline(
            profiles=profiles_stream,
            context=context,
            writer=writer,
            log=logger,
            on_profile_done=seen.add,
        )

        verified_path = os.path.join(output_dir, "verified_leads.csv")
        rejected_path = os.path.join(output_dir, "rejected_leads.csv")
        profiles_found = verified + rejected + skipped

        return {
            "profiles_found": profiles_found,
            "verified": verified,
            "rejected": rejected,
            "skipped": skipped,
            "verified_path": verified_path,
            "rejected_path": rejected_path,
            "verified_rows": _read_csv_rows(verified_path),
            "rejected_rows": _read_csv_rows(rejected_path),
        }
    finally:
        close_authenticated_context(context)
```

```python
import os
import sys

import click
from dotenv import load_dotenv

from src.runner import run_outreach

load_dotenv()


@click.command()
@click.option("--criteria", required=True, help='Search term, e.g. "co-founder" or "bits pilani"')
@click.option("--max-results", default=50, show_default=True, help="Target number of profiles")
@click.option(
    "--output-dir",
    default="output",
    show_default=True,
    help="Directory to write verified_leads.csv and rejected_leads.csv",
)
@click.option(
    "--cookies-file",
    default=os.getenv("LINKEDIN_COOKIES_FILE", "linkedin_cookies.json"),
    show_default=True,
    help="Path to LinkedIn session cookie file",
)
@click.option(
    "--seen-profiles-file",
    default=os.getenv("SEEN_PROFILES_FILE", "data/seen_profiles.txt"),
    show_default=True,
    help="Global dedup file — profiles already here are skipped across all runs",
)
def main(criteria, max_results, output_dir, cookies_file, seen_profiles_file):
    """
    Scrape LinkedIn for people matching CRITERIA, find and verify their emails,
    and save results to local CSV files. Skips anyone already in the seen-profiles
    file so repeated runs never re-process the same person.
    """
    result = run_outreach(
        criteria=criteria,
        max_results=max_results,
        output_dir=output_dir,
        cookies_file=cookies_file,
        seen_profiles_file=seen_profiles_file,
        log=click.echo,
    )

    if not result["profiles_found"]:
        click.echo("No new profiles found. Check your criteria or LinkedIn session.")
        sys.exit(0)

    click.echo("")
    click.echo("Done.")
    click.echo(f"  Verified  -> {output_dir}/verified_leads.csv  : {result['verified']}")
    click.echo(f"  Rejected  -> {output_dir}/rejected_leads.csv  : {result['rejected']}")
    click.echo(f"  Skipped   (errors)                           : {result['skipped']}")


if __name__ == "__main__":
    main()
```
